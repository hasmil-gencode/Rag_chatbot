# /etc/nginx/sites-available/genia
#
# Architecture:
# - Cloudflared -> Nginx :1224
# - Nginx -> Gateway :4000 for the main app and tenant API proxy
# - Nginx -> Tenant :1223 for embed assets/API that must be iframe-friendly
#
# Important for SSE/text streaming:
# - proxy_buffering off
# - proxy_cache off
# - gzip off
# - long proxy_read_timeout/proxy_send_timeout
#
# Security note:
# - /api/internal/ is blocked at Nginx before any proxy location.
# - Gateway can still call Tenant internal APIs directly via localhost/server URL.

upstream genia_gateway {
    server 127.0.0.1:4000;
    keepalive 32;
}

upstream genia_tenant {
    server 127.0.0.1:1223;
    keepalive 32;
}

server {
    listen 1224;
    server_name genia.gencode.com.my;

    client_max_body_size 100M;
    proxy_http_version 1.1;

    # Common proxy headers
    proxy_set_header Host $host;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto https;
    proxy_set_header X-Request-ID $request_id;

    # Block public access to internal tenant APIs.
    location ^~ /api/internal/ {
        return 404;
    }

    # External data ingest API direct to Tenant.
    # Used by GenForm/other systems with x-api-key + ingest:write scope.
    location = /api/v1/ingest/dynamic {
        proxy_pass http://genia_tenant;

        proxy_buffering off;
        proxy_cache off;
        proxy_read_timeout 300s;
        proxy_send_timeout 300s;
        proxy_connect_timeout 75s;
        proxy_request_buffering off;
    }

    # SSE browser chat stream via Gateway -> Tenant
    location = /api/chat/stream {
        proxy_pass http://genia_gateway;

        proxy_buffering off;
        proxy_cache off;
        gzip off;
        proxy_read_timeout 3600s;
        proxy_send_timeout 3600s;
        proxy_connect_timeout 75s;
        proxy_request_buffering off;

        add_header X-Accel-Buffering "no" always;
        add_header Cache-Control "no-cache, no-transform" always;
    }

    # SSE external API stream via Gateway x-api-key routing
    location = /api/v1/chat/stream {
        proxy_pass http://genia_gateway;

        proxy_buffering off;
        proxy_cache off;
        gzip off;
        proxy_read_timeout 3600s;
        proxy_send_timeout 3600s;
        proxy_connect_timeout 75s;
        proxy_request_buffering off;

        add_header X-Accel-Buffering "no" always;
        add_header Cache-Control "no-cache, no-transform" always;
    }

    # Embed widget routes direct to Tenant (allow iframe & cross-origin)
    location /embed/ {
        proxy_pass http://genia_tenant;

        proxy_buffering off;
        proxy_cache off;
        proxy_read_timeout 300s;
        proxy_send_timeout 300s;

        add_header X-Frame-Options "ALLOWALL" always;
        add_header Content-Security-Policy "frame-ancestors *;" always;
        add_header Access-Control-Allow-Origin "*" always;
        add_header Access-Control-Allow-Methods "GET, POST, OPTIONS" always;
        add_header Access-Control-Allow-Headers "Content-Type, Authorization, X-API-Key, X-Request-ID" always;

        proxy_hide_header Cross-Origin-Embedder-Policy;
        proxy_hide_header Cross-Origin-Opener-Policy;
        proxy_hide_header X-Frame-Options;
        proxy_hide_header Content-Security-Policy;
    }

    # Embed API direct to Tenant
    location /api/embed/ {
        proxy_pass http://genia_tenant;

        proxy_buffering off;
        proxy_cache off;
        proxy_read_timeout 300s;
        proxy_send_timeout 300s;

        add_header Access-Control-Allow-Origin "*" always;
        add_header Access-Control-Allow-Methods "GET, POST, OPTIONS" always;
        add_header Access-Control-Allow-Headers "Content-Type, Authorization, X-API-Key, X-Request-ID" always;

        if ($request_method = 'OPTIONS') {
            return 204;
        }
    }

    # Uploads: allow large request body, avoid request buffering surprises
    location = /api/upload {
        proxy_pass http://genia_gateway;

        proxy_request_buffering off;
        proxy_buffering off;
        proxy_cache off;
        proxy_read_timeout 900s;
        proxy_send_timeout 900s;
    }

    # Vite/React build assets are content-hashed. These can be cached long-term.
    # Keep this separate from the SPA HTML so users do not get a stale index.html
    # that points at deleted/new asset filenames after a deploy.
    location ^~ /assets/ {
        proxy_pass http://genia_gateway;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_cache_bypass $http_upgrade;
        proxy_read_timeout 300s;
        proxy_send_timeout 300s;
        proxy_connect_timeout 75s;

        add_header Cache-Control "public, max-age=31536000, immutable" always;
    }

    # The SPA entry document must be revalidated on every visit.
    location = /index.html {
        proxy_pass http://genia_gateway;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_cache_bypass $http_upgrade;
        proxy_read_timeout 300s;
        proxy_send_timeout 300s;
        proxy_connect_timeout 75s;

        add_header Cache-Control "no-cache, no-store, must-revalidate" always;
        add_header Pragma "no-cache" always;
        add_header Expires "0" always;
    }

    # Everything else through Gateway
    location / {
        add_header Strict-Transport-Security "max-age=31536000; includeSubDomains; preload" always;
        add_header Content-Security-Policy "default-src 'self'; script-src 'self' 'unsafe-inline' 'unsafe-eval' https://static.cloudflareinsights.com; style-src 'self' 'unsafe-inline'; img-src 'self' data: https:; font-src 'self' data:; connect-src 'self' https:; media-src 'self' blob:; frame-ancestors 'none';" always;
        add_header X-Content-Type-Options "nosniff" always;
        add_header X-Frame-Options "DENY" always;
        add_header Referrer-Policy "strict-origin-when-cross-origin" always;
        add_header Permissions-Policy "geolocation=(), microphone=(self), camera=()" always;
        add_header X-XSS-Protection "1; mode=block" always;

        proxy_pass http://genia_gateway;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_cache_bypass $http_upgrade;
        proxy_read_timeout 300s;
        proxy_send_timeout 300s;
        proxy_connect_timeout 75s;

        add_header Cache-Control "no-cache, no-store, must-revalidate" always;
        add_header Pragma "no-cache" always;
        add_header Expires "0" always;
    }
}

# After copying:
# sudo nginx -t
# sudo systemctl reload nginx
